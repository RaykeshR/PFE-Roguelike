# UTF-8
#
# A .rc file for PyInstaller to embed version info into the executable
#
VSVersionInfo(
    ffi=FixedFileInfo(
        # filevers and prodvers should be a tuple of four integers, (1, 2, 3, 4)
        # to correspond to the numeric version block in toplevel data
        filevers=(1, 0, 0, 0),
        prodvers=(1, 0, 0, 0),
        # Contains a bitmask that specifies the valid bits 'fileos'
        os=0x40004,
        # On Windows this should be root of the Unicode FPU codepage
        subtype=0x0,
        # Application type
        type=0x1,
    ),
    kids=[
        StringFileInfo(
            [
                StringTable(
                    u'040904B0',
                    [
                        StringStruct(u'CompanyName', u'PFE Roguelike Team'),
                        StringStruct(u'FileDescription', u'PFE Roguelike Game'),
                        StringStruct(u'FileVersion', u'1.0.0.0'),
                        StringStruct(u'InternalName', u'PFERoguelike'),
                        StringStruct(u'LegalCopyright', u'© 2025 PFE Roguelike Team. All rights reserved.'),
                        StringStruct(u'OriginalFilename', u'PFERoguelike.exe'),
                        StringStruct(u'ProductName', u'PFE Roguelike'),
                        StringStruct(u'ProductVersion', u'1.0.0.0'),
                    ]
                )
            ]
        ),
        VarFileInfo([VarStruct(u'Translation', [0x0409, 1200])])
    ]
)
